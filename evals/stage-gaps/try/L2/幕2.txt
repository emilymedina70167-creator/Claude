# L2 第 2 幕：行着看，写增广矩阵（三维平面没有，只试写矩阵这一半）
tex e = "\hl{r1}{x_1 + x_3 = 3} \quad \hl{r2}{x_2 + x_3 = 2}"
matrix T = 1 0 1 | 3 / 0 1 1 | 2
cover T
matrix M = 0 0 0 | 0 / 0 0 0 | 0
label lh M "x1   x2   x3  |  b" side=above tone=note
say "把 {e.r1}、{e.r2} 抄成一张数表，一列只放一个未知数的系数。"
open M edit range=-5..5
goal g1 "写出增广矩阵" when="M = T"
@g1 say "对，这就是增广矩阵。"
学生: write M.1,1 1
学生: write M.1,3 1
学生: write M.1,4 3
学生: write M.2,1 1
学生: write M.2,2 1
学生: write M.2,4 2
say "你写的第 2 行读回来是什么方程？"
ask say "第 2 行读成方程"
学生: say x1 + x2 = 2
say "嗯。"
