# L3 第 6 幕：特征网格、P⁻¹AP = Λ、三张卡排成 A
plane p range=-5..5
matrix A = 3 1 / 2 2
matrix P = 1 1 / 1 -2 beside=A
transform p P
vec p v = 2 -1 color=g1
say "这是用 ξ1、ξ2 当坐标轴的斜网格。{p.v} 在斜网格里的坐标是多少？"
ask next "看好了，按一下作用 A" button="作用 A"
学生: next
transform p A
say "{p.v} 变成了 Av。斜网格里看，A 干了什么？"
matrix L = 0 0 / 0 0
matrix Lt = 4 0 / 0 1
cover Lt
label ll L "Λ" side=left tone=key
open L edit range=-5..5
goal g1 "用斜网格当坐标，A 写成矩阵 Λ" when="L = Lt"
@g1 say "对，Λ = diag(4, 1)。"
@g1 array t = P⁻¹ Λ P
@g1 open t
@g1 ask next "把三张卡排成「□ □ □ = A」，排好点下面" button="排好了"
学生: write L.1,1 4
学生: write L.2,2 1
学生: swap t#1 t#3
学生: next
untransform p
transform p = 2/3 1/3 / 1/3 -1/3
transform p L
transform p P
say "照你排的顺序从右往左播了一遍。"
