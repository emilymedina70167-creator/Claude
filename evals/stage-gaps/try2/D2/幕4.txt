# D2 第 4 幕：红了好几个时只修离新结点最近的那个
# 第 1 幕插完 47 的树放回中间
tree t = 20 10 40 5 15 30 50 _ _ _ _ 25 35 45 _ _ _ _ _ _ 47
bf t
# 每个结点下面的子树高度、50 那个位置的高度尺：没有指令；只能 tag 手写死字
tag t.50 "高3·前2"
say "红了两个。先点你打算先修哪个，再动手。"
ask tap t.20 t.50 "先修哪个？点它" expect=t.50
学生: tap t.20
# 老师接：交给学生拎，走偏马上叫老师（让后果说话）
open t rotate wrong=quiet call=wrong undo=on
say "好，就修 {t.20}。动手拎。"
@t.done glow t
@t.done say "平了。"
学生: rotate t.20 left
# 老师接：指着还红的 50
pulse t.50
say "你修的是谁？还红的是谁？"
ask tap t "还红的是谁？点它" expect=t.50
学生: tap t.50
# 「回到插入后」：没有这个按钮。先试学生「撤一步」（老师接过一轮以后还撤不撤得了）
say "按「撤一步」回去，这次修 {t.50}。"
ask next "撤好了点下面" button="撤好了"
学生: undo
学生: next
# 老师接：撤不了，只能老师把插入后的树整棵重造一遍
tree t = 20 10 40 5 15 30 50 _ _ _ _ 25 35 45 _ _ _ _ _ _ 47
bf t
tag t.50 "高3·前2"
open t rotate wrong=quiet undo=on
say "回到插入后了。这次修 {t.50}。"
@t.done glow t
@t.done say "平了。"
@t.done ask say "20 你一下都没碰，它怎么也好了？"
学生: rotate t.45 left
学生: rotate t.50 right
学生: say 50那棵转完又变回原来的高度了，上面就跟没插一样
# 老师接：高度尺（没有）；两个按钮 + 理由
card k = "一处" "可能好几处"
say "那插一个数，最多要修几处？"
ask tap k "点一个" expect=k.1
学生: tap k.1
ask say "为什么？写一句"
学生: say 修好的那棵子树高度回到插入前，上面的平衡因子都不变
