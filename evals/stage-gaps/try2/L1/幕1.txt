# L1 第 1 幕：收尾 AB = O（造 D ≠ C 让 AD = AC），再翻成 BA、DA、CA
matrix A = 1 2 / 2 4
matrix B = 2 2 / -1 -1 beside=A
matrix C = 1 0 / 2 1 beside=B
label tb B "你上次造的" side=above tone=note
product P = A B
mul P
# D 想要一张空牌
matrix D = _ _ / _ _
matrix D = 0 0 / 0 0
matrix AC = {A C}
matrix AD = {A D} beside=AC
label lac AC "AC" side=left tone=key
label lad AD "AD" side=left tone=key
say "上次你造的 {B} 让 {P} = O。先收个尾：造一个不等于 {C} 的 {D}，让 {AD} = {AC}。"
open D edit range=-3..3
goal g1 "改 D 的数：AD = AC，D 却不是 C" when="AD = AC, D != C"
@g1 say "对，AD = AC，D 却不是 C。"
@g1 card q = "BA 也是 O" "BA 不是 O" "说不准"
@g1 ask tap q "现在把 A、B 换个位置乘。先猜：BA 也是 O 吗？"
# 学生一格一格乱试（每按一下 ± 算一下）
学生: write D.2,2 1
学生: write D.2,2 0
学生: write D.1,1 1
学生: write D.2,1 2
学生: write D.1,1 3
学生: write D.1,2 1
# 页面叫老师（试了好多下）：老师指着 AB = O 那张
glow P
say "被 A 乘完是 O 的，你手上有谁？"
学生: write D.1,2 2
学生: write D.2,1 1
学生: tap q.2
# 猜完，老师翻牌：BA、DA、CA
unglow P
matrix P = {B A}
matrix AC = {C A}
matrix AD = {D A}
retext lac "CA"
retext lad "DA"
say "换了顺序，{P} 不是 O 了；{AD}、{AC} 也一起翻了过来。"
say "这节课就问一件事：顺序换了，什么时候结果不变？"
ask tap AD "{AD} 和 {AC} 哪几格不一样？点一格" expect=AD.1,2
学生: tap AD.1,2
say "嗯。"
