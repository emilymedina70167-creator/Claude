# L1 第 3 幕：练习 2.1 学生自己搭 (A−B)(A+B) 的方块（拿两个拨数代替拖边沿标签），−BA 和 AB 抵消不掉
card q = "练习 2.1：A、B 如下，A² − B² 和 (A−B)(A+B) 分别是？" "(A) 两个都是 (7 5; −4 −1)" "(B) 两个都是 (−4 −2; 8 10)" "(C) A² − B² = (7 5; −4 −1)，(A−B)(A+B) = (−4 −2; 8 10)" "(D) A² − B² = (−4 −2; 8 10)，(A−B)(A+B) = (7 5; −4 −1)"
matrix A = 3 4 / -1 -2
matrix B = 1 1 / -3 -2 beside=A
dial s = 1 range=-1..1 at=A
dial t = 1 range=-1..1 at=B
label ls dial:s "左边括号：A + {s}·B" side=right
label lt dial:t "右边括号：A + {t}·B" side=right
matrix AA = {A A}
matrix AB = {t A B} beside=AA
matrix BA = {s B A}
matrix BB = {s t B B} beside=BA
label c1 AA "A·A" side=left tone=note
label c2 AB "A·(tB)" side=right tone=note
label c3 BA "(sB)·A" side=left tone=note
label c4 BB "(sB)·(tB)" side=right tone=note
matrix Tot = {AA + AB + BA + BB}
matrix Dir = {(A - B)(A + B)} beside=Tot
label lto Tot "四块合计" side=above tone=note
label ldi Dir "(A−B)(A+B) 直接乘" side=above tone=note
cover Dir
say "这题你自己来：拨两个括号里 B 前面的号，搭出 (A−B)(A+B)。"
goal g1 "搭出 (A−B)(A+B)：合计和直接乘的叠上" when="Tot = Dir"
@g1 uncover Dir
@g1 say "合计和直接乘的叠上了。"
@g1 ask say "数里 −ab 和 ba 会消掉。这儿 {AB} 和 {BA} 消得掉吗？"
学生: dial t -1
学生: dial s -1
学生: dial t 1
学生: say 消不掉，它们不是相反的
matrix Mid = {AB + BA}
label lm Mid "中间两块合起来" side=above tone=key
matrix D2 = {A^2 - B^2} beside=Mid
label ld2 D2 "A² − B²" side=above tone=key
say "{AB} 和 {BA} 合起来剩下 {Mid}，没消掉。"
ask tap q "选哪一项？" expect=q.4
学生: tap q.4
say "两个结果为什么不一样？"
ask say "指一块给我看，写一句为什么"
学生: say 因为 AB ≠ BA，中间剩下 AB − BA
say "嗯。"
