# L2 分支 2：做「r2 − r1」时把第 1 行也改了
matrix M0 = 1 2 0 | 2 / 1 0 -4 | 0 / 1 2 -2 | 0
label l0 M0 "上一张" side=above tone=note
matrix M1 = 1 2 0 | 2 / 1 0 -4 | 0 / 1 2 -2 | 0 beside=M0
label l1 M1 "这一步写在这里" side=above tone=note
say "做 r2 − r1，把新的整张写进 {M1}。"
open M1 edit input=keypad
ask value "这一步做了什么？"
学生: write M1.1,1 0
学生: write M1.1,3 4
学生: write M1.2,1 0
学生: write M1.2,2 -2
学生: write M1.2,4 -2
学生: value r2 - r1
# 老师：哪几行变了（两张相减）、P 代进去、正确的那张并排
matrix Dlt = {M1 - M0}
label ldl Dlt "你这张 − 上一张：哪几行变了" side=above tone=key
matrix Pv = 4 / -1 / 1 / -1
matrix Res = {M1 Pv} beside=Pv
label lr Res "P 代进你这张" side=above tone=note
matrix E21 = 1 0 0 / -1 1 0 / 0 0 1
matrix Ok = {E21 M0} beside=E21
label lok Ok "只改第 2 行的结果" side=above tone=note
dim E21
say "看 {Dlt}：哪几行变了？P 代进你这张，第 1 行还是 0 吗？"
ask say "被加的那一行变，拿来加的那一行呢？"
学生: say 拿来加的那行不该变
# 用把手重做一次：先把 M1 写回上一张
write M1.1,1 1
write M1.1,3 0
write M1.2,1 1
write M1.2,2 0
write M1.2,4 0
close M1
open M1 rowop wrong=mark undo=on
say "用把手做一次：把第 1 行拖到第 2 行上。"
goal g1 "用把手做 r2 − r1" when="M1 = Ok"
@g1 say "第 1 行一动没动。"
学生: rowop M1 r2 - r1
say "嗯。"
