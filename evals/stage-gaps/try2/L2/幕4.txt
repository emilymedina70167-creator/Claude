# L2 第 4 幕：学生自己一步一张消成阶梯形（先用数字键盘整行写，再用行的把手），从下往上读出 P（三维的平面、P 要三维）
matrix M0 = 1 2 0 | 2 / 1 0 -4 | 0 / 1 2 -2 | 0
label l0 M0 "上一张" side=above tone=note
matrix M1 = 1 2 0 | 2 / 1 0 -4 | 0 / 1 2 -2 | 0 beside=M0
label l1 M1 "这一步写在这里" side=above tone=note
say "把台阶线下面全变成 0，一步写一张。先做第一步，把新的整张写进 {M1}。"
open M1 edit input=keypad
ask value "这一步做了什么？（比如 r2 − r1）"
学生: write M1.2,1 0
学生: write M1.2,2 -2
学生: write M1.2,3 4
学生: write M1.2,4 -2
学生: value r2 - r1
# 老师看出第 2 行第 3 格算错了：把 P 代进上一张和这一张
matrix Pv = 4 / -1 / 1 / -1
matrix R0 = {M0 Pv}
matrix R1 = {M1 Pv} beside=R0
label lr0 R0 "P 代进上一张" side=above tone=note
label lr1 R1 "P 代进你这张" side=above tone=note
glow M1.r2
say "把 P 代进去：你这张第 2 行不是 0 了。P 还在你这一行上吗？"
goal g1 "改对这一行：和上一张同解" when="M1 same M0"
@g1 say "对，P 又回来了。"
学生: write M1.2,3 -4
# 剩下的交给行的把手（页面判每一下）
remove R0
remove R1
remove Pv
unglow M1
close M1
open M1 rowop wrong=mark undo=on
say "剩下的用把手做：拖一行的把手到另一行上。"
goal g2 "消成阶梯形" when="M1 stairs"
@M1.done say "阶梯形 ✓"
@M1.done stairs M1
@M1.done ask value "最下面一行里只剩谁？x3 = ？" answer=1
学生: rowop M1 r3 - 2r1
学生: undo
学生: rowop M1 r3 - r1
学生: value 1
ask value "往上一行：x2 = ？" answer=-1 wrong="1=符号错了; -3=把 −4x3 当成 +4x3"
学生: value -3
say "把 x3 = 1 代进 {M1.r2} 再算一遍。"
ask value "x2 = ？" answer=-1
学生: value -1
ask value "x1 = ？" answer=4
学生: value 4
say "台阶一共几级？"
ask value "台阶几级？" answer=3
学生: value 3
say "嗯。"
