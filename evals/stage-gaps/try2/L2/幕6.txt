# L2 第 6 幕：自由变量、特解、导出组、通解（Q 沿 L 拖、L 平移成 L0、点沿 L 跑要三维；这里量矩阵和代回去的那一半）
matrix M = 1 2 -2 | 0 / 0 -2 -2 | -2 / 0 0 0 | 0
stairs M
glow M.c3
# 代回去要系数矩阵和右边分开（增广矩阵不能直接拿来乘）：另摆两张，灰着
matrix A0 = 1 2 -2 / 0 -2 -2 / 0 0 0
matrix bb = 0 / -2 / 0 beside=A0
dim A0 bb
matrix Eta = 0 / 0 / 0
matrix R1 = {A0 Eta - bb} beside=Eta
label le Eta "x3 = 0 时的点（你来写）" side=above tone=note
label lr1 R1 "代回去：左边 − 右边" side=above tone=note
say "第 3 列没有台阶头，x3 你说了算。让 x3 = 0，解出 x1、x2，写进 {Eta}。"
open Eta edit input=keypad
goal g1 "写出 x3 = 0 时的解" when="R1 = O, Eta.3,1 = 0"
@g1 say "对，代回去全是 0。"
学生: write Eta.1,1 2
学生: write Eta.2,1 1
学生: write Eta.1,1 -2
# 第二个点 x3 = 1，再连一根箭头（两点之差跟着算）
matrix Q1 = 0 / 0 / 1
matrix R2 = {A0 Q1 - bb} beside=Q1
matrix Df = {Q1 - Eta}
label lq Q1 "x3 = 1 时的点（你来写）" side=above tone=note
label lr2 R2 "代回去" side=above tone=note
label ld Df "两点连成的箭头" side=left tone=key
say "再让 x3 = 1，写进 {Q1}。看两点连成的箭头 {Df}。"
open Q1 edit input=keypad
goal g2 "写出 x3 = 1 时的解" when="R2 = O, Q1.3,1 = 1"
@g2 say "对。"
学生: write Q1.1,1 2
学生: write Q1.2,1 0
# 把右边慢慢换成 0：右边那一格跟着拨数
remove R2
dial c = -2 range=-2..0 at=M
matrix M = 1 2 -2 | 0 / 0 -2 -2 | {c} / 0 0 0 | 0
matrix bb = 0 / {c} / 0 beside=A0
say "把右边全换成 0：拨 {c} 到 0。右边是 0 的方程组里，x3 = 1 时解是什么？写进 {Xi}。"
matrix Xi = 0 / 0 / 1
matrix R3 = {A0 Xi} beside=Xi
label lx Xi "右边换成 0 以后，x3 = 1（你来写）" side=above tone=note
label lr3 R3 "代进 Ax = 0" side=above tone=note
open Xi edit input=keypad
goal g3 "写出 Ax = 0 在 x3 = 1 时的解" when="R3 = O, Xi != O"
@g3 say "对，它和那根箭头一样。"
学生: dial c 0
学生: write Xi.1,1 2
学生: write Xi.2,1 0
# 学生把 (2, 0, 1) 当成 ξ（带着右边那一列算的）：拿拨数 k 扫一遍它写的通解，代回去看
dial k = 0 range=-3..3 at=Xi
matrix X = {Eta + k Xi}
matrix Rk = {A0 X - bb} beside=X
label lxk X "η + k·你的 ξ" side=above tone=key
label lrk Rk "代回去" side=above tone=note
set c -2
say "右边先换回去。拨 {k}，看 η + k·你的 ξ 代回去还是不是 0。"
ask next "拨几下 k，看完点下面" button="看完了"
学生: dial k 1
学生: next
say "k = 1 就不是解了。你的 ξ 是在哪个方程组里取的？"
ask say "为什么？"
学生: say 我用了右边是 -2 的那组，应该用右边是 0 的
set c 0
say "在右边是 0 的那组里重新取 x3 = 1。"
goal g4 "改对 ξ" when="R3 = O, Xi != O"
@g4 say "对，ξ = (4, −1, 1)，和箭头一样。"
学生: write Xi.1,1 4
学生: write Xi.2,1 -1
ask say "L 上随便一个点，怎么一句话写出来？"
学生: say x = (−2, 1, 0) + k(4, −1, 1)
say "嗯。"
