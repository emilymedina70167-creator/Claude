# D1 第 3 幕：d 缩小到 1，叫出名字；试验台
array a = 13 27 49′ 55 04 49 38 65 97 76
array b = 13 04 49′ 38 27 49 55 65 97 76
label lb b "第 1 幕的下排" side=left tone=soft
dim b
dial d = 5 range=1-9 at=a
label dl dial:d "d = {d}" side=right
group a stride=d
card q = "趟数记录（d｜这一趟后｜挪几格｜累计）" "5｜13 27 49′ 55 04 49 38 65 97 76｜5｜5"
say "再来一趟，这回隔 3。"
# 学生自己拨 d（这一次工具能真拨）；拨到 3 页面替老师分泳道、交出去；相等的两张 49 用 <= 写
goal g0 "把 $d$ 拨到 3" when="d = 3"
@g0 split a
@g0 open a undo=on
@g0 goal g1 "三组各自排好" when="a.1 <= a.4, a.4 <= a.7, a.7 <= a.10, a.2 <= a.5, a.5 <= a.8, a.3 <= a.6, a.6 <= a.9"
@g1 close a
@g1 merge a
@g1 ask say "认得这一排吗？还差什么才算排好？"
学生: dial d 3
学生: swap a#7 a#4
学生: swap a#5 a#2
学生: say 跟第一幕的下排一样，还要d=1再走一遍
# 学生把整排拖到下排上叠着比：一排格子没有整排拖的动作。老师写 compare 试试；「排好了」按钮配 on= 的挑战
undim b
compare a b
addline q "3｜13 04 49′ 38 27 49 55 65 97 76｜2｜7"
say "和 {b} 一模一样。"
goal g3 "这一排排好了" when="a sorted" on="排好了"
ask next "觉得排好了就点；没排好就接着拨 $d$" button="排好了"
学生: next
say "还没排好。哪一趟能保证全排好？拨拨看。"
goal g4 "把 $d$ 拨到那一趟" when="d = 1"
学生: dial d 1
# d=1：页面不会照直接插入自动走，老师一张张 move，挪几格自己数
