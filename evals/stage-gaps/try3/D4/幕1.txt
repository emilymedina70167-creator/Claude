# D4 第 1 幕：拖卡对调，让 49 左边全比它小、右边全不比它小；「灯」= 左上角的挑战
# 「49 左右」不按格子写死：49 在第 k 格的 8 种情况用 | 隔开（部件比部件）
array a = 49 38 65 97 76 13 27 49′
paint a#1 g4
paint a#8 g1
array s = 13 27 38 49 49′ 65 76 97 hide=on
label goal a "49 左边全比 49 小，右边全不比 49 小" side=above tone=key
say "挪一挪，让 {a#1} 左边全比它小、右边全不比它小。别的不用排好。"
open a undo=on
goal g1 "49 左边全比它小、右边全不比它小" when="a.1 = a#1, a.2-8 >= a#1 | a.1 < a#1, a.2 = a#1, a.3-8 >= a#1 | a.1-2 < a#1, a.3 = a#1, a.4-8 >= a#1 | a.1-3 < a#1, a.4 = a#1, a.5-8 >= a#1 | a.1-4 < a#1, a.5 = a#1, a.6-8 >= a#1 | a.1-5 < a#1, a.6 = a#1, a.7-8 >= a#1 | a.1-6 < a#1, a.7 = a#1, a.8 >= a#1 | a.1-7 < a#1, a.8 = a#1"
@g1 keep a "站定时"
@g1 label ok a#1 "站定" side=below tone=key
@g1 lift a#1
@g1 say "对，49 站住了。"
@g1 ask value "你排的和别人的可能不一样。{a#1} 站在第几格？" answer=4
