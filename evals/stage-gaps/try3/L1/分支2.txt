# 配方台：x²、−5x 换成两张方牌，+3 是一枚圆币（1×1 的一张）
matrix T1 = {A^2}
matrix T2 = {-5 A} beside=T1
matrix c = 3 beside=T2
label l1 T1 "x² → A²" side=above tone=note
label l2 T2 "−5x → −5A" side=above tone=note
label lc c "+3（一个数）" side=above tone=note
matrix Sum = collect
label ls Sum "合计" side=above tone=key
open T1 put
open T2 put
open c put
say "把三项一项项放进合计。"
ask next "放完点下面" button="放完了"
学生: put T1 Sum
学生: put T2 Sum
学生: put c Sum
学生: next
matrix Th = 3 3 / 3 3
matrix TE = {3 E} beside=Th
label lth Th "你每格加 3：偷偷换成的" side=above tone=note
label lte TE "3E" side=above tone=note
say "数加不到 {T2} 这种表上。你每格加 3，等于把 3 换成了 {Th}。"
ask say "f(x) 里的 3 是 3·x⁰，x⁰ 到了 A 这边是谁？"
学生: say E
open TE put
say "那把 {TE} 放进合计。"
ask next "放完点下面" button="放完了"
学生: put TE Sum
学生: next
matrix W = 0 3 / 3 0

label lwd Wd "多出来的" side=above tone=note
say "3 只铺在对角线上，合计是 {Sum}。你写的多出来 {Wd}。"
