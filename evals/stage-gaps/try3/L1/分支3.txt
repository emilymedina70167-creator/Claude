# L1 分支 3（第三次）：拿「位置可换」当理由；E 碰到了不会消（−2AE − EA 并成 −AE）；(A+E)³ 硬乘带着 E
card k = "依据卡：" "结合律（挪括号）" "数乘结合律（圆币滑动）" "EC = C" "CE = C" "交换（指出哪两颗）"
tex g = "(5\hl{e}{E})\hl{c}{C} = \hl{c2}{C}(5\hl{e2}{E})"
say "证 {g}。第一步凭哪张卡？"
ask tap k "第一步凭哪张卡？" expect=k.2-5
学生: tap k.6
# 拿「交换」卡：同一张卡也能「证」DC = CD（页面算两边）
matrix D = 1 0 / 0 2
matrix C = 0 1 / 0 0 beside=D
matrix DC = {D C}
matrix CD = {C D} beside=DC
label ldc DC "DC" side=above tone=key
label lcd CD "CD" side=above tone=key
glow k.6 g
compare DC CD
say "你拿的卡就是要证的那句。同一张卡也能「证」{DC} = {CD}，可它们不一样。"
ask say "那这张卡能当理由吗？"
学生: say 不能
say "那重走一遍：链里用了哪几张卡？都点出来。"
ask pick k "(5E)C = 5(EC) = 5C = 5(CE) = C(5E)，用了哪几张？" button="点好了" expect="k.3 k.4 k.5"
学生: pick k.3 k.4
# E 不会消：学生写 −2AE − EA = −AE
remove D
remove C
remove DC
remove CD
matrix A = 2 -1 / -3 3
matrix X = {-2 A E}
matrix Y = {-(E A)} beside=X
label lx X "−2AE" side=above tone=note
label ly Y "−EA" side=above tone=note
say "你写的是 −2AE − EA = −AE。把 {X} 拖到 {Y} 上并一下。"
open X merge
ask next "并完点下面" button="并好了"
学生: merge X Y
学生: next
matrix AE = {A E}
matrix EA = {E A} beside=AE
label lae AE "AE" side=above tone=note
label lea EA "EA" side=above tone=note
compare AE A
say "先看 {AE}、{EA}，和 {A} 比。"
matrix Mine = {-(A E)}
matrix Right = {-2 A E - E A} beside=Mine
matrix Gap = {Right - Mine}
label lm Mine "你写的 −AE" side=above tone=note
label lr Right "−2AE − EA" side=above tone=key
label lg Gap "差" side=left tone=note
say "你的 {Mine} 和 {Right} 差了 {Gap}。"
@ask.right say "对，−3A。"
ask value "−2AE − EA 是 A 的几倍？" answer=-3
学生: value -1
say "AE 就是 A，EA 也是 A。"
ask value "−2AE − EA 是 A 的几倍？" answer=-3
学生: value -3
# (A+E)³ 硬乘：学生写的 EA² = 4AE，代具体的 A 两边各算
remove X
remove Y
remove Mine
remove Right
remove Gap
remove AE
remove EA
matrix A = 1 1 / 1 1
matrix S1 = {E A^2}
matrix S2 = {4 A E} beside=S1
label ls1 S1 "EA²" side=above tone=note
label ls2 S2 "你写的 4AE" side=above tone=note
compare S1 S2
say "代 A = {A}：{S1} 和 {S2} 一样吗？带着的 E 都是谁？"
ask say "EA² 化到最简是什么？"
学生: say 2A
say "嗯。"
