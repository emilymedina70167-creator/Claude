# L1 第 1 幕（第三次）：收尾 AB = O（造 D ≠ C 让 AD = AC：先拨格子，再把牌放进 D），再翻成 BA、DA、CA
matrix A = 1 2 / 2 4
matrix B = 2 2 / -1 -1 beside=A
matrix C = 1 0 / 2 1 beside=B
label tb B "你上次造的" side=above tone=note
product P = A B
mul P
# D 想要一张空牌：空牌上挂不上跟着算的 AD，退回 0
matrix D = _ _ / _ _
matrix AC = {A C}
matrix AD = {A D} beside=AC
matrix D = 0 0 / 0 0
matrix AD = {A D} beside=AC
label lac AC "AC" side=left tone=key
label lad AD "AD" side=left tone=key
# 老师要的事实：D − C、A(D − C)（藏着，现状照写）
matrix DmC = {D - C} hide=on
matrix AK = {A DmC} hide=on
say "上次你造的 {B} 让 {P} = O。先收个尾：造一个不等于 {C} 的 {D}，让 {AD} = {AC}。"
open D edit range=-3..3
open AD compare
# 一格一句：停一次就知道哪几格已经一样了
goal g1 "改 D 的数：AD = AC，D 却不是 C" when="AD.1,1 = AC.1,1, AD.1,2 = AC.1,2, AD.2,1 = AC.2,1, AD.2,2 = AC.2,2, D != C"
@g1 say "对，AD = AC，D 却不是 C。"
@g1 card q = "BA 也是 O" "BA 不是 O" "说不准"
@g1 ask tap q "现在把 A、B 换个位置乘。先猜：BA 也是 O 吗？"
# 学生一格一格乱试（每按一下 ± 算一下）
学生: write D.1,1 1
学生: write D.2,1 2
学生: write D.2,2 1
学生: compare AD AC
学生: write D.1,1 2
学生: write D.2,2 0
学生: write D.1,1 3
# 页面叫老师（试了好多下）：老师指着 AB = O 那张，再把 D 换成能往里放牌的
glow P
matrix D = collect
open B put
open C put
say "被 A 乘完是 O 的，你手上有谁？把牌放进 {D} 试试。"
学生: put C D
学生: put B D
学生: tap q.2
# 猜完，老师翻牌：A、B 换槽成 BA；AC、AD 换成 CA、DA
unglow P
product P = B A
mul P
remove AC
remove AD
matrix CA = {C A}
matrix DA = {D A} beside=CA
label lca CA "CA" side=left tone=key
label lda DA "DA" side=left tone=key
matrix Gap = {DA - CA} hide=on
say "换了顺序，{P} 不是 O 了；{DA}、{CA} 也一起翻了过来。"
say "这节课就问一件事：顺序换了，什么时候结果不变？"
ask pick DA "{DA} 和 {CA} 哪几格不一样？都点出来" button="点好了" expect=DA
学生: pick DA.1,2 DA.2,1
say "嗯。"
