matrix A = 2 -1 / -3 3
matrix B1 = {A A}

# 并不上（页面不认 A·(−2E) 是 A 的倍数）：老师先把每块化开（依据卡没有，老师替学生化），再交给学生并
matrix B2 = {-2 A}
matrix B3 = {-A}
matrix B4 = {2 E}
matrix B1 = {A^2}
retext c2 "−2A"
retext c3 "−A"
retext c4 "2E"
retext c1 "A²"
open B1 merge
open B2 merge
say "化开了。现在再并：只有都是 A 的几倍才并得上。"
matrix M3 = {-3 A} hide=on

@g1 say "并上了：−3A。"
学生: merge B1 B3
学生: merge B2 B3

label lt Tot "合计" side=above tone=key
close B1

open B3 put
open B4 put

@g2 show Dir
@g2 label ld Dir "(A−E)(A−2E) 直接乘" side=above tone=key
@g2 link x.p B1 g1
@g2 link x.q B3 g2
@g2 link x.s B4 g4
