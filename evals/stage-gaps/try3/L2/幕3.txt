# L2 第 3 幕（第三次）：学生先在空格子里写增广矩阵（页面读回方程）；倍加 r3 + k·r1 拨着看（格里写式子）；再真的拖把手做（三维的平面、铰链要三维）
tex e = "\hl{e1}{x_1 + 2x_2 = 2},\quad \hl{e2}{x_1 - 4x_3 = 0},\quad \hl{e3}{x_1 + 2x_2 - 2x_3 = 0}"
matrix T = 1 2 0 | 2 / 1 0 -4 | 0 / 1 2 -2 | 0 hide=on
matrix M = _ _ _ | _ / _ _ _ | _ / _ _ _ | _ vars="x1 x2 x3"
label lh M "x1    x2    x3  |  b" side=above tone=note
say "先把三个方程抄成增广矩阵：一列只放一个未知数的系数。"
open M edit input=keypad
goal g1 "写出增广矩阵" when="M same T"
@g1 say "对，写对了。"
学生: write M.1,1 1
学生: write M.1,2 2
学生: write M.1,3 0
学生: write M.1,4 2
学生: write M.2,1 1
学生: write M.2,2 0
学生: write M.2,3 -4
学生: write M.2,4 0
# 写到第 8 格，页面叫老师（试了好多下）
say "慢慢写，写满了页面替你对。"
学生: write M.3,1 1
学生: write M.3,2 2
学生: write M.3,3 -2
学生: write M.3,4 0
# 拨 k：第 3 行的格里直接写「原来的 + k × 第 1 行」，第 1、2 行照抄 M
close M
dial k = 0 range=-3..3 at=N
matrix N = {M.1,1} {M.1,2} {M.1,3} | {M.1,4} / {M.2,1} {M.2,2} {M.2,3} | {M.2,4} / {M.3,1 + k M.1,1} {M.3,2 + k M.1,2} {M.3,3 + k M.1,3} | {M.3,4 + k M.1,4} vars="x1 x2 x3" beside=M
label ln N "r3 + k·r1 以后" side=above tone=key
matrix Pv = 4 / -1 / 1
label lp Pv "P = (4, −1, 1)" side=above tone=note
matrix Res = {coef(N) Pv - rhs(N)} hide=on
say "把 {M.r1} 的 k 倍加到 {M.r3} 上。拨 k，盯着：谁在动、谁不动、{Pv} 走没走。"
goal g2 "拨到第 3 行第 1 个数变成 0" when="N.3,1 = 0"
@g2 say "做到了。"
@g2 ask say "拨的时候什么没动？第 3 个方程现在是什么样？"
学生: dial k 2
学生: dial k -1
学生: say 第 1、2 行没动，P 一直满足，第 3 行变成 −2x3 = −2
# 再交给学生真的做一次：拖第 1 行的把手到第 3 行上、写倍数
remove N
remove Pv
remove Res
unlabel dial:k
keep M "倍加之前"
open M rowop wrong=quiet undo=on
open M back="倍加之前"
say "这回你自己在 {M} 上做：把第 1 行的把手拖到第 3 行上，写倍数。"
goal g3 "让第 3 行第 1 个数变成 0" when="M.3,1 = 0"
@g3 say "做到了：这叫倍加。"
学生: rowop M r3 + 2r1
学生: undo
学生: rowop M r3 - r1
say "嗯。"
